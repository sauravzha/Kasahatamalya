    <script>
      function scrollLeaderTrack(direction) {
        const track = document.getElementById('leaderTrack');
        if (!track) return;
        // Scroll width for 3 cards
        const cardWidth = track.querySelector('.leader-card-item')?.offsetWidth || 340;
        const scrollAmount = (cardWidth + 32) * 3 * direction;
        track.scrollBy({ left: scrollAmount, behavior: 'smooth' });
      }

      function jumpLeaderSlide(index) {
        const track = document.getElementById('leaderTrack');
        if (!track) return;
        const cardWidth = track.querySelector('.leader-card-item')?.offsetWidth || 340;
        const scrollTarget = (cardWidth + 32) * 3 * index;
        track.scrollTo({ left: scrollTarget, behavior: 'smooth' });
        updateDots(index);
      }

      function updateDots(activeIndex) {
        const dots = document.querySelectorAll('#leaderDots .leader-dot');
        dots.forEach((dot, idx) => {
          if (idx === activeIndex) dot.classList.add('active');
          else dot.classList.remove('active');
        });
      }

      // Sync dots on user scroll
      document.addEventListener('DOMContentLoaded', () => {
        const track = document.getElementById('leaderTrack');
        if (track) {
          track.addEventListener('scroll', () => {
            const cardWidth = track.querySelector('.leader-card-item')?.offsetWidth || 340;
            const step = (cardWidth + 32) * 3;
            const currentIndex = Math.round(track.scrollLeft / step);
            updateDots(Math.min(currentIndex, 4));
          });
        }
      });

      function filterLeaders(category, btnEl) {
        const buttons = document.querySelectorAll('.leader-filters .filter-btn');
        buttons.forEach(b => b.classList.remove('active'));
        if (btnEl) btnEl.classList.add('active');

        const items = document.querySelectorAll('#leaderTrack .leader-card-item');
        items.forEach(item => {
          if (category === 'all' || item.getAttribute('data-category') === category) {
            item.style.display = 'block';
          } else {
            item.style.display = 'none';
          }
        });
        
        const track = document.getElementById('leaderTrack');
        if (track) track.scrollTo({ left: 0, behavior: 'smooth' });
      }
